# Служебные эндпоинты и загруженные картинки индексировать незачем.
#
# Исключение — две картинки превью ссылки (og:image). Они генерятся на лету и
# поэтому лежат под /api/, а не среди статики. Без явного Allow их не заберут
# краулеры, которые robots.txt уважают: Googlebot-Image и Twitterbot. Telegram
# и Discord его игнорируют — из-за этого поломка НЕ видна при обычной проверке
# «кинул ссылку себе в мессенджер»: там карточка с картинкой, а в Twitter и в
# выдаче Google та же ссылка остаётся без неё. Allow длиннее и специфичнее
# Disallow ниже, поэтому выигрывает по правилу самого длинного совпадения.
Allow: /api/og-tierlist.php
Allow: /api/og-news.php
Disallow: /api/
Disallow: /images/

# Админка — внутренние страницы (/admin — редактор тирлиста, /admin/news —
# новости, /admin/promo — реклама). Разметку они отдают только по паролю, эта
# строка лишь убирает их из выдачи; префикс закрывает и вложенные адреса.
# Медиакит на сайте не лежит вовсе — он в docs/, вне веб-корня.
Disallow: /admin
